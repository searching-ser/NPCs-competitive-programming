\section{Li Chao Tree / Convex Hull Trick}
\noindent{\scriptsize
\textbf{Convex Hull Trick (CHT) Optimization:}\\
Maintains lower envelope of lines $y = mx + c$ in $O(\log X)$. Optimizes:
\[ dp[i] = \min_{j < i} (m_j \cdot x_i + c_j) + \text{extra}(i) \]
\textbf{Quadratic Cost (e.g. Frog 3):} $dp[i] = \min_j \{dp[j] + (h_i - h_j)^2 + C\}$
\[ \implies m = -2h_j,\quad c = dp[j] + h_j^2,\quad x = h_i,\quad \text{extra} = h_i^2 + C \]
}
\vspace{0.3em}
